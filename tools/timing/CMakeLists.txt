cmake_minimum_required(VERSION 3.16)
project(merlin_timing VERSION 0.1.0 LANGUAGES CXX)

add_library(merlin_timing INTERFACE)
target_compile_features(merlin_timing INTERFACE cxx_std_17)
target_include_directories(merlin_timing INTERFACE
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)

include(CTest)
if(BUILD_TESTING)
  add_executable(merlin_timing_test tests/test_engine.cpp)
  target_link_libraries(merlin_timing_test PRIVATE merlin_timing)
  add_test(NAME merlin_timing_core COMMAND merlin_timing_test)
  add_executable(merlin_timing_reservations_test tests/test_reservations.cpp)
  target_link_libraries(merlin_timing_reservations_test PRIVATE merlin_timing)
  add_test(NAME merlin_timing_reservations COMMAND merlin_timing_reservations_test)
  add_executable(merlin_timing_instruction_context_test tests/test_instruction_context.cpp)
  target_link_libraries(merlin_timing_instruction_context_test PRIVATE merlin_timing)
  add_test(NAME merlin_timing_instruction_context COMMAND merlin_timing_instruction_context_test)
  add_executable(merlin_timing_instruction_suffix_test tests/test_instruction_suffix.cpp)
  target_link_libraries(merlin_timing_instruction_suffix_test PRIVATE merlin_timing)
  add_test(NAME merlin_timing_instruction_suffix COMMAND merlin_timing_instruction_suffix_test)
  add_executable(merlin_timing_memory_milestones_test tests/test_memory_milestones.cpp)
  target_link_libraries(merlin_timing_memory_milestones_test PRIVATE merlin_timing)
  add_test(NAME merlin_timing_memory_milestones COMMAND merlin_timing_memory_milestones_test)
  add_executable(merlin_timing_posted_timeline_test tests/test_posted_timeline.cpp)
  target_link_libraries(merlin_timing_posted_timeline_test PRIVATE merlin_timing)
  add_test(NAME merlin_timing_posted_timeline COMMAND merlin_timing_posted_timeline_test)
  add_executable(merlin_timing_posted_engine_stage_test tests/test_posted_engine_stage.cpp)
  target_link_libraries(merlin_timing_posted_engine_stage_test PRIVATE merlin_timing)
  add_test(NAME merlin_timing_posted_engine_stage COMMAND merlin_timing_posted_engine_stage_test)
endif()

install(DIRECTORY include/ DESTINATION include)
install(TARGETS merlin_timing EXPORT merlin_timing_targets)
install(EXPORT merlin_timing_targets NAMESPACE merlin_timing:: DESTINATION lib/cmake/merlin_timing)
